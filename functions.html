Question 2

Swap numbers using Call by address.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void swap(int *a, int *b)
{
int t;
t=*a;
*a=*b;
*b=t;
}

void main()
{
int x,y;
clrscr();

printf("Enter two numbers: ");
scanf("%d%d",&x,&y);

swap(&x,&y);

printf("After swapping: %d %d",x,y);

getch();
}

Output

Enter two numbers: 10 20
After swapping: 20 10

{

Armstrong Numbers using local and global variable

#include<conio.h>
#include<dos.h>

{

scanf("%d",&n);

{
r=n%10;

else
printf("Not an Armstrong number");
}

{
clrscr();
armstrong();
getch();
}

Enter a number: 153
Armstrong number
