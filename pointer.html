Question 1

A pointer should point the array & through pointer we will fill all the locations of the array by assigning the value from the user & print the array.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
int arr[10], *ptr, i;
clsrcr();
for(i=0; i<=9; i++)
{
ptr=&arr[i];
scanf("%d",ptr);
printf("value of index %d=%d\n",i,*ptr);
}
delay(3000);
}

Output

10,20,30,40
value of index 0=10
value of index 1=20
value of index 2=30
value of index 3=40


Question 2

Write a program in which we are having 2D array of 2 by 3. Insert the value through pointer & print the values.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
int arr[2][3], *ptr, i,j;
clsrcr();
ptr=arr[0];
for(i=0; i<=1; i++)
{
	for(j=0; j<=2; j++)
	{
	printf("Enter the value:");
	scanf("%d",&arr[i][j]);
	}
}
for(i=0; i<=1; i++)
{
	for(j=0; j<=2; j++)
	{
	printf("%u",*ptr);
	ptr++;
	}
printf("\n");
}
delay(3000);
}

Output

Enter the value:10,20,30,40,50,60
10		20		30
40		50		60
