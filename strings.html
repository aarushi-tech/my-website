Question 1

Enter a string from user and convert in uppercase.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
int main()
{
char str[100];
int i;
clrscr();

printf("Enter a string: ");
fgets(str, sizeof(str), stdin);

for (i = 0; str[i] != '\0'; i++)
{
    if (str[i] >= 'a' && str[i] <= 'z')
    {
        str[i] = str[i] - 32;
    }
}
printf("Uppercase string: %s", str);
delay(3000);
}

Output

Enter a string: Hello World
Uppercase string: HELLO WORLD

Question 18

String palindrome example (using reverse).

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
char s1[100],s2[100];
int i=0,j;
clrscr();

printf("Enter a string: ");
gets(s1);

while(s1[i]!='\0')
{
i++;
}

for(j=0;i>0;j++)
{
i--;
s2[j]=s1[i];
}

s2[j]='\0';

printf("Reverse string = %s\n",s2);

i=0;
while(s1[i]!='\0' && s1[i]==s2[i])
{
i++;
}

if(s1[i]=='\0' && s2[i]=='\0')
printf("String is palindrome");
else
printf("String is not palindrome");

getch();
}

Output

Enter a string: madam
Reverse string = madam
String is palindrome
