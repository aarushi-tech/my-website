Question 1

Write a program of recursion and display the output.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
int x=1;
void fun()
{
if(x==5)
{
printf("%d",x);
x++;
fun();
}
printf("A");
}
void main()
{
clrscr();
fun();
delay(3000);
}

Output

12345AAAAAA
