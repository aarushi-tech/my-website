Question 19

Enter password from user maximum 20 digits / character press enter key if you want to exit.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
    char pass[21];
    int i=0;
    char ch;

    clrscr();

    printf("Enter password (max 20 characters): ");

    while(i<20)
    {
        ch=getch();

        if(ch==13)
            break;

        pass[i]=ch;
        printf("*");
        i++;
    }

    pass[i]='\0';

    printf("\nPassword entered.");

    delay(4000);
}

Output

Enter password (max 20 characters): ******
Password entered.

Question 23

Enter a number from user and check it is Armstrong number or not.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
    int n, temp, r, sum=0;
    clrscr();

    printf("Enter a number: ");
    scanf("%d",&n);

    temp=n;

    while(n>0)
    {
        r=n%10;
        sum=sum+(r*r*r);
        n=n/10;
    }

    if(temp==sum)
        printf("Armstrong number");
    else
        printf("Not an Armstrong number");

    delay(5000);
}

Output

Enter a number: 153
Armstrong number
