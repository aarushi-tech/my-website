Question 1

Enter the four number from user and print it into matrix.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
    int a[2][2], i, j;

    clrscr();

    printf("Enter four numbers:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("\nMatrix is:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d\t", a[i][j]);
        }
        printf("\n");
    }

    delay(4000);
}

Output

Enter four numbers:
1
2
3
4

Matrix is:
1       2
3       4

Question 3

Enter 2 matrix from user and add them.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
    int a[10][10], b[10][10], c[10][10];
    int i, j, r, col;

    clrscr();

    printf("Enter number of rows and columns: ");
    scanf("%d%d", &r, &col);

    printf("\nEnter elements of first matrix:\n");
    for(i = 0; i < r; i++)
    {
        for(j = 0; j < col; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("\nEnter elements of second matrix:\n");
    for(i = 0; i < r; i++)
    {
        for(j = 0; j < col; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    /* Add the two matrices */
    for(i = 0; i < r; i++)
    {
        for(j = 0; j < col; j++)
        {
            c[i][j] = a[i][j] + b[i][j];
        }
    }

    printf("\nAddition of two matrices:\n");
    for(i = 0; i < r; i++)
    {
        for(j = 0; j < col; j++)
        {
            printf("%d\t", c[i][j]);
        }
        printf("\n");
    }

    delay(4000);
}

Output

Enter number of rows and columns: 2 2

Enter elements of first matrix:
1 2
3 4

Enter elements of second matrix:
5 6
7 8

Addition of two matrices:
6       8
10      12

#include<dos.h>
void main()
{
    int a[10][10], b[10][10], c[10][10];

    {

    for(i = 0; i < r; i++)
    {
        for(j = 0; j < col; j++)

Question 5

Enter 2 matrix from user and multiply them.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
    int a[10][10], b[10][10], c[10][10];
    int i, j, k, r1, c1, r2, c2;

    clrscr();

    printf("Enter rows and columns of first matrix: ");
    scanf("%d%d", &r1, &c1);

    printf("Enter rows and columns of second matrix: ");
    scanf("%d%d", &r2, &c2);

    if(c1 != r2)
    {
        printf("Matrix multiplication is not possible.");
        getch();
        return;
    }

    printf("\nEnter elements of first matrix:\n");
    for(i = 0; i < r1; i++)
    {
        for(j = 0; j < c1; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("\nEnter elements of second matrix:\n");
    for(i = 0; i < r2; i++)
    {
        for(j = 0; j < c2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    /* Initialize result matrix */
    for(i = 0; i < r1; i++)
    {
        for(j = 0; j < c2; j++)
        {
            c[i][j] = 0;

            for(k = 0; k < c1; k++)
            {
                c[i][j] = c[i][j] +
                          a[i][k] * b[k][j];
            }
        }
    }

    printf("\nMultiplication of two matrices:\n");
    for(i = 0; i < r1; i++)
    {
        for(j = 0; j < c2; j++)
        {
            printf("%d\t", c[i][j]);
        }
        printf("\n");
    }

    delay(4000);
}

Output

Enter rows and columns of first matrix: 2 2
Enter rows and columns of second matrix: 2 2

Enter elements of first matrix:
1 2
3 4

Enter elements of second matrix:
5 6
7 8

Multiplication of two matrices:
19      22
43      50
