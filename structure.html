Question 1

Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
struct employee
{
    int ID;
    char name[30];
    char designation[30];
};

void main()
{
    clrscr();

    struct employee emp[50];
    int i;

    printf("Enter details of 50 employees:\n");

    for(i = 0; i < 50; i++)
    {
        printf("\nEmployee %d\n", i + 1);

        printf("Enter Employee ID: ");
        scanf("%d", &emp[i].ID);

        printf("Enter Name: ");
        scanf("%s", emp[i].name);

        printf("Enter Designation: ");
        scanf("%s", emp[i].designation);
    }

    printf("\n\n");
    printf("--------------------------------------------------\n");
    printf("\n","ID", "Name", "Designation");
    printf("--------------------------------------------------\n");

    for(i = 0; i < 50; i++)
    {
        printf("%-10d %-20s %-20s\n",
               emp[i].ID,
               emp[i].name,
               emp[i].designation);
    }

    printf("--------------------------------------------------");

    delay(4000);
}
