Question 1

Enter an array from the user and display all the value.

#include<dos.h>

{

Question 2

Enter an array from user and print the array in reverse order.

#include<conio.h>
#include<dos.h>
int main() 
{
int n, i;
clrscr();

printf("Enter the size of array: ");
scanf("%d", &n);

int arr[n];

printf("Enter array elements:\n");
for (i = 0; i <  n; i++) 
	{
        scanf("%d", &arr[i]);
    }

printf("Array in reverse order:\n");
for (i = n - 1; i >= 0; i--) 
	{
        printf("%d ", arr[i]);
    }

delay(3000);
}

Enter array elements:
10 20 30 40 50
Array in reverse order:
50 40 30 20 10

Question 10

#include<conio.h>
#include<dos.h>

{

clrscr();

printf("Enter 5 array elements:\n");
for (i = 0; i < 5; i++) 
	{
        scanf("%d", &arr[i]);
    }

    printf("Enter value to search: ");
    scanf("%d", &num);

    for (i = 0; i < 5; i++) 
	{
        if (arr[i] == num) 
		{
            found = 1;
            break;
        }
    }

    if (found == 1) 
	{
        printf("Found");
    } 
	else 
	{
        printf("Not found");
    }
delay(3000);
}

Output

10 20 30 40 50
Enter value to search: 30
Found


Question 11

Enter five values of array and sort in ascending order.

#include<stdio.h>
#include<conio.h>
#include<dos.h>
int main() 
{
int arr[5], i, j, temp;
clrscr();

printf("Enter 5 array elements:\n");
for (i = 0; i < 5; i++) 
	{
        scanf("%d", &arr[i]);
    }

    for (i = 0; i < 5; i++) 
	{
        for (j = i + 1; j < 5; j++) 
		{
            if (arr[i] > arr[j]) 
			{
                temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
    }

printf("Array in ascending order:\n");
for (i = 0; i < 5; i++) 
	{
        printf("%d ", arr[i]);
    }
delay(3000);
}

Output

Enter 5 array elements:
50 20 40 10 30
Array in ascending order:
10 20 30 40 50
